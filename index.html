    Google Apps Script Integration

    Kode Code.gs & Setup Spreadsheet

    Salin kode di bawah ini ke editor Apps Script Google Sheets Anda untuk database online.

    Panduan Hubungkan Web Ke Google Sheets:

    1. Buat Google Spreadsheet baru di Google Drive Anda.
    2. Klik menu Ekstensi > Apps Script.
    3. Hapus semua kode bawaan, lalu Paste (Tempel) kode `Code.gs` di bawah ini.
    4. Jalankan fungsi `setupDatabase()` sekali untuk otomatis membuat sheet & tabel.
    5. Klik Deploy > Web App (Execute as: Me, Who has access: Anyone).
    Code.gs Google Apps Script 
    // ==========================================
// CODE.GS - BACKEND SMAN 7 PONTIANAK DATABASE
// ==========================================

function setupDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1. Sheet Pengaturan General
  var pengSheet = ss.getSheetByName("Pengaturan") || ss.insertSheet("Pengaturan");
  if (pengSheet.getLastRow() === 0) {
    pengSheet.appendRow(["Key", "Value"]);
    pengSheet.appendRow(["heroTitle", "Mewujudkan Generasi Unggul, Berkarakter & Berdaya Saing Global"]);
    pengSheet.appendRow(["kepsekNama", "Drs. H. M. Yasin, M.Pd"]);
    pengSheet.appendRow(["kepsekNip", "NIP. 19680312 199403 1 005"]);
  }

  // 2. Sheet Berita
  var beritaSheet = ss.getSheetByName("Berita") || ss.insertSheet("Berita");
  if (beritaSheet.getLastRow() === 0) {
    beritaSheet.appendRow(["ID", "Judul", "Kategori", "Tanggal", "Ringkasan", "Gambar"]);
  }

  // 3. Sheet Galeri Foto
  var galeriSheet = ss.getSheetByName("Galeri") || ss.insertSheet("Galeri");
  if (galeriSheet.getLastRow() === 0) {
    galeriSheet.appendRow(["ID", "Judul", "Kategori", "URL Gambar", "Caption"]);
  }

  // 4. Sheet Ekstrakurikuler
  var ekskulSheet = ss.getSheetByName("Ekskul") || ss.insertSheet("Ekskul");
  if (ekskulSheet.getLastRow() === 0) {
    ekskulSheet.appendRow(["ID", "Nama", "Kategori", "Deskripsi"]);
  }

  // 5. Sheet Guru
  var guruSheet = ss.getSheetByName("Guru") || ss.insertSheet("Guru");
  if (guruSheet.getLastRow() === 0) {
    guruSheet.appendRow(["ID", "Nama", "Jabatan", "Mapel", "Foto"]);
  }

  // 6. Sheet Buku Tamu
  var guestSheet = ss.getSheetByName("BukuTamu") || ss.insertSheet("BukuTamu");
  if (guestSheet.getLastRow() === 0) {
    guestSheet.appendRow(["ID", "Tanggal", "Nama", "Peran", "Kontak", "Pesan"]);
  }

  return "Setup Database SMAN 7 Pontianak Berhasil!";
}

// Simpan Buku Tamu Baru
function addBukuTamu(nama, peran, kontak, pesan) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("BukuTamu");
  sheet.appendRow([Date.now(), new Date().toLocaleDateString(), nama, peran, kontak, pesan]);
  return { success: true };
}

// Simpan Foto Galeri Baru
function addFotoGaleri(judul, kategori, url, caption) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Galeri");
  sheet.appendRow([Date.now(), judul, kategori, url, caption]);
  return { success: true };
}
